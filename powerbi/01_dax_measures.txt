Total Patients = DISTINCTCOUNT(Dim_Patient[patient_id])
Total Encounters = DISTINCTCOUNT(Fact_Encounter[encounter_id])
Completed Encounters = CALCULATE([Total Encounters], Fact_Encounter[status] = "Completed")
Total Encounter Cost = SUM(Fact_Encounter[encounter_cost])
Average Cost per Encounter = DIVIDE([Total Encounter Cost], [Total Encounters])
Average LOS = AVERAGE(Fact_Encounter[length_of_stay_days])
Total Appointments = DISTINCTCOUNT(Fact_Appointment[appointment_id])
No Show Appointments = CALCULATE([Total Appointments], Fact_Appointment[status] = "No Show")
No Show Rate % = DIVIDE([No Show Appointments], [Total Appointments])
Cancelled Appointments = CALCULATE([Total Appointments], Fact_Appointment[status] = "Cancelled")
Cancellation Rate % = DIVIDE([Cancelled Appointments], [Total Appointments])
Average Wait Minutes = AVERAGE(Fact_Encounter[wait_minutes])
Readmission Encounters = CALCULATE([Total Encounters], Fact_Encounter[readmission_flag] = "Yes")
Readmission Rate % = DIVIDE([Readmission Encounters], [Total Encounters])
High Cost Encounters = CALCULATE([Total Encounters], Fact_Encounter[high_cost_flag] = "High")
High Cost Rate % = DIVIDE([High Cost Encounters], [Total Encounters])
